  let cal=new Date(2025,4,1),marks=new Set();
  function drawCal(){
    const y=cal.getFullYear(),m=cal.getMonth();$("#month").textContent=`${y}年${m+1}月⌄`;
    const g=$("#grid");g.innerHTML="";const first=new Date(y,m,1).getDay(),n=new Date(y,m+1,0).getDate();

    for(let d=1;d<=n;d++){const key=`${y}-${m+1}-${d}`,e=document.createElement("div");e.className="day";const hh=(d*3)%9;
      e.innerHTML=`<div class="box ${marks.has(key)?"marked":""}"><div class="fill" style="height:${Math.min(100,hh/8*100)}%"></div></div><div class="num">${d}</div>`;
      e.addEventListener("click",()=>{marks.has(key)?marks.delete(key):marks.add(key);e.querySelector(".box").classList.toggle("marked")});
      g.appendChild(e);
    }
  }
